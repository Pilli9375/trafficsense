% LaTeX preamble of the report (scripts/build_report.py, pandoc --include-in-header)
\usepackage{fontspec}
\setmainfont{texgyretermes}[Extension=.otf, UprightFont=*-regular, BoldFont=*-bold,
  ItalicFont=*-italic, BoldItalicFont=*-bolditalic]
\setmonofont{texgyrecursor}[Extension=.otf, UprightFont=*-regular, BoldFont=*-bold,
  ItalicFont=*-italic, BoldItalicFont=*-bolditalic, Scale=0.95]
\usepackage{etoolbox}
\usepackage[font=small, labelfont=bf, labelsep=colon]{caption}
% Tables: smaller type, tighter rows, single spacing
\AtBeginEnvironment{longtable}{\setstretch{1.0}\setlength{\tabcolsep}{3.5pt}}
\usepackage{setspace}
% Chapter headings: "Chapter N" above the title, less vertical space than the report default
\usepackage{titlesec}
\titleformat{\chapter}[display]{\normalfont\Large\bfseries\centering}{\chaptertitlename\ \thechapter}{8pt}{\LARGE}
\titlespacing*{\chapter}{0pt}{-20pt}{24pt}
% Allow long `code` names and URLs to break
\usepackage{xurl}
\setlength{\emergencystretch}{3em}
\sloppy
% Floats: keep figures near their text
\renewcommand{\topfraction}{0.9}\renewcommand{\textfraction}{0.08}\renewcommand{\floatpagefraction}{0.8}
\usepackage{float}
\let\origfigure\figure
\let\endorigfigure\endfigure
\renewenvironment{figure}[1][]{\origfigure[htbp]}{\endorigfigure}
% Code listings wrap instead of running into the margin
\usepackage{fvextra}
\AtBeginDocument{%
  \RecustomVerbatimEnvironment{Highlighting}{Verbatim}{commandchars=\\\{\},breaklines,breakanywhere,fontsize=\footnotesize}%
  \renewenvironment{verbatim}{\VerbatimEnvironment\begin{Verbatim}[breaklines,breakanywhere,fontsize=\footnotesize]}{\end{Verbatim}}%
}
% Very wide tables go on landscape pages (rotated in PDF viewers)
\usepackage{pdflscape}
\usepackage{needspace}
